-- QCC Expansión Tiendas Mercadona Online -- El año se pasa como variable {{anio}} (en Metabase: tipo Número, p. ej. 2027). -- -- IMPORTANTE: el filtro de año va al final, en el WHERE de la consulta -- principal, y NO dentro de los CTE. Las funciones de ventana necesitan -- ver el histórico completo de cada centro para calcular prev_model, -- prev_pom y prev_online. Si se filtrara antes, el primer movimiento de -- cada año se interpretaría como alta nueva en lugar de como cambio. WITH ordered AS ( SELECT cv.code, cv.mercadona_code, cv.start_date, cv.end_date, cv.model, cv.pom, cv.online, LAST_VALUE(cv.model IGNORE NULLS) OVER ( PARTITION BY cv.code ORDER BY cv.start_date ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING ) AS prev_model, LAST_VALUE(cv.pom IGNORE NULLS) OVER ( PARTITION BY cv.code ORDER BY cv.start_date ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING ) AS prev_pom, LAG(cv.online) OVER ( PARTITION BY cv.code ORDER BY cv.start_date ) AS prev_online FROM `centrosmo_mart.center_versions` cv ), with_nums AS ( SELECT *, CAST(REGEXP_EXTRACT(CAST(model AS STRING), r'\d+') AS INT64) AS model_num, CAST(REGEXP_EXTRACT(CAST(prev_model AS STRING), r'\d+') AS INT64) AS prev_model_num FROM ordered ), postal_codes AS ( SELECT code, start_date, STRING_AGG(postal_code, ' ' ORDER BY postal_code) AS cps_que_servimos FROM `centrosmo_mart.center_versions_postal_codes` WHERE service_type <> 'nsd' GROUP BY code, start_date ) SELECT ci.zone AS zona_tiendas, w.code AS num_centro, ci.name AS denominacion, UPPER(ci.town) AS poblacion, UPPER(ci.province) AS provincia, CASE WHEN w.online = FALSE AND (w.prev_online = TRUE OR w.prev_online IS NULL) THEN '' WHEN w.online = TRUE AND w.prev_online = FALSE THEN CONCAT('Apertura tras reforma NTC', CAST(w.model_num AS STRING)) WHEN w.prev_model IS NULL AND w.online = TRUE THEN CONCAT('NTC', CAST(w.model_num AS STRING)) WHEN w.model_num > w.prev_model_num THEN CONCAT('Ampliación de NTC', CAST(w.prev_model_num AS STRING), ' a NTC', CAST(w.model_num AS STRING)) WHEN w.model_num < w.prev_model_num THEN CONCAT('Bajada de NTC', CAST(w.prev_model_num AS STRING), ' a NTC', CAST(w.model_num AS STRING)) ELSE 'Cambio lateral' END AS escenario, pc.cps_que_servimos AS cps_que_servimos, NULL AS tiendas_que_dejan_de_servir_tc, NULL AS fecha_1er_dia_formacion, CASE WHEN w.online = FALSE AND (w.prev_online = TRUE OR w.prev_online IS NULL) THEN w.start_date END AS fecha_cierre_servicio, CASE WHEN w.online = TRUE THEN w.start_date END AS fecha_1er_servicio FROM with_nums w LEFT JOIN `centrosmo_mart.center_info` ci ON w.code = ci.code LEFT JOIN postal_codes pc ON w.code = pc.code AND w.start_date = pc.start_date WHERE ( w.prev_model IS NULL OR w.model <> w.prev_model OR w.pom <> w.prev_pom OR w.online <> w.prev_online ) AND EXTRACT(YEAR FROM w.start_date) = {{anio}} AND w.code NOT IN ('vlc1','bcn1','mad1','svq1','alc1','mad2','mad3','pan1','pan2','pan3','mlg1','pmi1') ORDER BY w.start_date, w.code
